      does not break View.get_value(pin).

Heavy lifting (ROUTER bind, View Tracker registration, heartbeat scan)

import inspect
import logging
import time
from dataclasses import dataclass
from typing import Any, Callable, Dict, Optional, Tuple

logger = logging.getLogger(__name__)

_STALE_POLICIES = ("hold_last", "return_default", "return_none")

# MessagePack primitives — the ONLY types a @signal may declare for v1.
# Richer payloads must go through @event(payload={...}).
_ALLOWED_DTYPES: Tuple[type, ...] = (int, float, bool, str)

# Readiness-gate timeout bounds. wait_timeout_s MUST be in [5, 600]; None is rejected.
# A 600s ceiling is a foot-gun guard — researchers can't accidentally configure a
# multi-hour wait. The mics_task readiness gate sums to max(wait_timeout_s) across
# all required sources, so this is a per-source ceiling, not a task ceiling.
_WAIT_TIMEOUT_MIN_S = 5
_WAIT_TIMEOUT_MAX_S = 600
_WAIT_TIMEOUT_DEFAULT_S = 60


def _resolve_signal_dtype(fn: Callable, default: Any) -> type:
    """Resolve a signal's dtype at class-build time.

    Order: (1) method's return annotation; (2) type(default).
    Raises TypeError if neither yields a type in _ALLOWED_DTYPES — loud failure
    before any data flows over the wire.
    """
    ann = fn.__annotations__.get("return")
    if isinstance(ann, type) and ann in _ALLOWED_DTYPES:
        return ann
    if default is not None and type(default) in _ALLOWED_DTYPES:
        return type(default)
    raise TypeError(
        f"@signal {fn.__name__}: must declare a value type via a return "
        f"annotation (e.g. '-> float') or a typed default (e.g. default=0.0). "
        f"Allowed dtypes: {[t.__name__ for t in _ALLOWED_DTYPES]}. "
        f"Got annotation={ann!r}, default={default!r}."
    )


def _resolve_command_signature(fn: Callable) -> Tuple[Tuple[Tuple[str, Optional[type]], ...], Optional[type]]:
    """Capture (args, returns) annotations for a @command method.

    Returned types are bare Python types where available (UI may inspect them);
    unannotated params surface as (name, None). Self is excluded.
    """
    sig = inspect.signature(fn)
    annotations = fn.__annotations__
    args: list = []
    for pname, p in sig.parameters.items():
        if pname == "self":
            continue
        if p.kind in (inspect.Parameter.VAR_POSITIONAL, inspect.Parameter.VAR_KEYWORD):
            continue
        ann = annotations.get(pname)
        args.append((pname, ann if isinstance(ann, type) else None))
    ret = annotations.get("return")
    return tuple(args), (ret if isinstance(ret, type) else None)

@dataclass(frozen=True)
class _SignalSpec:
    name: str
    default: Any
    stale_after_ms: int
    stale_policy: str  # one of _STALE_POLICIES
    dtype: type        # one of _ALLOWED_DTYPES — resolved at class-build time


@dataclass(frozen=True)
class _EventSpec:
    name: str
    payload: Dict[str, type]


@dataclass(frozen=True)
class _CommandSpec:
    name: str
    args: Tuple[Tuple[str, Optional[type]], ...]   # ((arg_name, type_or_None), ...)
    returns: Optional[type]


def signal(default: Any = None, stale_after_ms: int = 200,
           stale_policy: str = "hold_last") -> Callable:
    """Mark a method as a streamed signal. The method body is never executed;
    the decorator only records the wire-spec.

    The signal's value dtype is inferred at class-build time from:
      1. the method's return annotation (e.g. `def x(self) -> float: ...`), or
      2. the type of `default` (e.g. `default=0.0` → float).
    Raises TypeError immediately if neither resolves to one of int/float/bool/str.

    stale_policy: "hold_last" | "return_default" | "return_none"
    """
    if stale_policy not in _STALE_POLICIES:
        raise ValueError(
            f"stale_policy must be one of {_STALE_POLICIES}, got {stale_policy!r}"
        )

    def decorator(fn: Callable) -> Callable:
        dtype = _resolve_signal_dtype(fn, default)
        # If default was provided but mismatches the dtype, coerce now (loud, build-time).
        coerced_default = default
        if default is not None and not isinstance(default, dtype):
            try:
                coerced_default = dtype(default)
            except (TypeError, ValueError) as exc:
                raise TypeError(
                    f"@signal {fn.__name__}: default {default!r} not coercible "
                    f"to declared dtype {dtype.__name__}: {exc}"
                ) from exc
        fn._extlink_signal = _SignalSpec(
            name=fn.__name__,
            default=coerced_default,
            stale_after_ms=stale_after_ms,
            stale_policy=stale_policy,
            dtype=dtype,
        )
        return fn

    return decorator


def event(payload: Optional[Dict[str, type]] = None) -> Callable:
    """Mark a method as a streamed event with an optional payload shape."""
    def decorator(fn: Callable) -> Callable:
        fn._extlink_event = _EventSpec(name=fn.__name__, payload=payload or {})
        return fn

    return decorator


def command(fn: Callable) -> Callable:
    """Mark a method as a command sendable Pi → SDK. Captures the method's
    parameter and return annotations so the state-builder UI (Phase 19+) can
    render a typed call form."""
    args, returns = _resolve_command_signature(fn)
    fn._extlink_command = _CommandSpec(name=fn.__name__, args=args, returns=returns)
    return fn


# ---------------------------------------------------------------- metaclass
class _Meta(type):
    """Walk the MRO at class-build time and harvest decorator metadata into
    class-level lookup tables. Subclasses inherit + extend their parents'
    declarations."""

    def __new__(mcs, name, bases, ns):
        cls = super().__new__(mcs, name, bases, ns)
        signals: Dict[str, _SignalSpec] = {}
        events: Dict[str, _EventSpec] = {}
        commands: Dict[str, _CommandSpec] = {}

        for klass in reversed(cls.__mro__):
            for attr in vars(klass).values():
                spec = getattr(attr, "_extlink_signal", None)
                if isinstance(spec, _SignalSpec):
                    signals[spec.name] = spec
                spec = getattr(attr, "_extlink_event", None)
                if isinstance(spec, _EventSpec):
                    events[spec.name] = spec
                spec = getattr(attr, "_extlink_command", None)
                if isinstance(spec, _CommandSpec):
                    commands[spec.name] = spec

        cls._declared_signals = signals
        cls._declared_events = events
        cls._declared_commands = commands
        return cls


# ---------------------------------------------------------------- base class
class ExternalHardware(metaclass=_Meta):
    """Base class for external-data hardware modules.

    Subclass and decorate methods with @signal / @event / @command. Instances
    are created by the base init_hardware path (task.py:162-217) using the
    per-pilot config from pilot_hardware_config.config.

    The __init__ stores the config but does NOT bind a socket. mics_task
    init_hardware calls .bind(ioloop, view) after super().init_hardware()
    returns; that is where the ROUTER is opened, View Trackers are
    registered, and the heartbeat scan starts.
    """

    # Class-level — populated by _Meta
    _declared_signals: Dict[str, _SignalSpec] = {}
    _declared_events: Dict[str, _EventSpec] = {}
    _declared_commands: Dict[str, _CommandSpec] = {}

    # Conformance to the existing base init_hardware path:
    is_trigger = False

    def __init__(self, *,
                 listen_port: int,
                 source_id: str,
                 stale_ms: int = 3000,
                 required: bool = True,
                 wait_timeout_s: int = _WAIT_TIMEOUT_DEFAULT_S,
                 name: Optional[str] = None,
                 event_dispatcher=None,
                 pi=None,         # ignored — base init_hardware passes it
                 run_id=None,     # ignored — base init_hardware passes it
                 class_name: Optional[str] = None,  # absorbed from config
                 type: Optional[str] = None,        # absorbed from config
                 **kwargs):
        if not source_id:
            raise ValueError(f"{type or self.__class__.__name__}: source_id required")
        if not isinstance(listen_port, int) or listen_port <= 0:
            raise ValueError(f"{source_id}: listen_port must be a positive int")

        # Readiness-gate config. wait_timeout_s == None is REJECTED — tasks can
        # never silently wait forever. Out-of-range is also rejected (loud).
        if wait_timeout_s is None:
            raise ValueError(
                f"{source_id}: wait_timeout_s must be an int in "
                f"[{_WAIT_TIMEOUT_MIN_S}, {_WAIT_TIMEOUT_MAX_S}], got None. "
                f"'Wait forever' is not allowed — the task could hang indefinitely."
            )
        if not isinstance(wait_timeout_s, int) or not (
            _WAIT_TIMEOUT_MIN_S <= wait_timeout_s <= _WAIT_TIMEOUT_MAX_S
        ):
            raise ValueError(
                f"{source_id}: wait_timeout_s must be an int in "
                f"[{_WAIT_TIMEOUT_MIN_S}, {_WAIT_TIMEOUT_MAX_S}], got {wait_timeout_s!r}"
            )

        self.source_id = source_id
        self.listen_port = int(listen_port)
        self.stale_ms = int(stale_ms)
        self.required = bool(required)
        self.wait_timeout_s = int(wait_timeout_s)
        self.name = name or source_id
        self.event_dispatcher = event_dispatcher

        # latest_values[sig_name] = (value, ts_pi_recv_seconds)
        self._latest: Dict[str, Any] = {
            sig: (spec.default, 0.0)
            for sig, spec in self._declared_signals.items()
        }
        self._signal_trackers: Dict[str, Any] = {}
        self._alive_tracker = None
        self._socket = None     # populated by .bind()
        self._view = None       # populated by .bind()

    # ------------------------------------------------ View / init_hardware contract
    def get_state(self):
        """Required by View.get_value(pin) when task.py:204 sets
        self.view.view[pin] = hw. Returns None — FDA reads use per-signal
        Trackers (e.g. dlc_cam1.left_paw_x), not the slot itself."""
        return None

    @property
    def alive(self) -> bool:
        """Read-only liveness flag for the mics_task readiness gate.
        False before .bind() runs and whenever the heartbeat scan has marked
        the source stale. The same value is mirrored in the
        f'{source_id}.alive' Boolean_Tracker for FDA transition use."""
        return bool(self._socket is not None and self._socket.alive)

    # ------------------------------------------------ lifecycle
    def bind(self, ioloop, view) -> None:
        """Open the ROUTER socket, register View Trackers, start heartbeat.
        Idempotent — calling twice is a no-op after the first."""
        if self._socket is not None:
            logger.debug("%s: bind() called twice — ignored", self.source_id)
            return

        self._view = view

        # Register one Tracker per declared signal + an alive Boolean_Tracker.
        for sig, spec in self._declared_signals.items():
            tracker_name = f"{self.source_id}.{sig}"
            self._signal_trackers[sig] = view.add_Tracker(
                info_type=tracker_name, initial_value=spec.default
            )
        self._alive_tracker = view.add_BooleanTracker(
            info_type=f"{self.source_id}.alive", initial_value=False
        )

        # Create + start the private socket helper (defined in Task 2).
        self._socket = _ExternalSocket(
            owner=self,
            ioloop=ioloop,
            event_dispatcher=self.event_dispatcher,
            listen_port=self.listen_port,
            source_id=self.source_id,
            stale_ms=self.stale_ms,
        )
        self._socket.start()
        logger.info("ExternalHardware[%s] bound on port %d", self.source_id,
                    self.listen_port)

    def close(self) -> None:
        if self._socket is not None:
            try:
                self._socket.close()
            except Exception:
                logger.exception("ExternalHardware[%s]: socket.close failed",
                                 self.source_id)
            finally:
                self._socket = None

    # ------------------------------------------------ inbound hooks (called by _ExternalSocket)
    def _on_signal_update(self, sig_name: str, value: Any,
                          ts_pi_recv: float) -> None:
        if sig_name not in self._declared_signals:
            return  # _ExternalSocket already logged + dropped
        self._latest[sig_name] = (value, ts_pi_recv)
        tracker = self._signal_trackers.get(sig_name)
        if tracker is None:
            return
        try:
            tracker.set(value)
        except Exception:
            logger.exception("%s.%s tracker.set failed", self.source_id, sig_name)

    def _on_event(self, evt_name: str, payload: Dict[str, Any],
                  ts_pi_recv: float) -> None:
        # Authors may override in subclasses; default no-op (the EVT is
        # already pushed to Event_Dispatcher by _ExternalSocket).
        pass

    def _on_stale_transition(self, alive: bool) -> None:
        """Called by _ExternalSocket heartbeat scan on alive↔stale flips."""
        if self._alive_tracker is not None:
            try:
                self._alive_tracker.set(alive)
            except Exception:
                logger.exception("%s.alive tracker.set failed", self.source_id)
        if not alive:
            # Apply per-signal stale policy to trackers
            for sig_name, spec in self._declared_signals.items():
                if spec.stale_policy == "hold_last":
                    continue
                tracker = self._signal_trackers.get(sig_name)
                if tracker is None:
                    continue
                fallback = (spec.default if spec.stale_policy == "return_default"
                            else None)
                try:
                    tracker.set(fallback)
                except Exception:
                    logger.exception("%s.%s stale-policy push failed",
                                     self.source_id, sig_name)

    # ------------------------------------------------ stale-aware read (ad-hoc)
    def get_signal(self, sig_name: str) -> Any:
        """Direct read with stale-policy applied. FDA transitions normally use
        view.get_value(<source_id>.<sig_name>) which already reflects the
        policy via _on_stale_transition. This is for ad-hoc callers."""
        spec = self._declared_signals.get(sig_name)
        if spec is None:
            raise KeyError(sig_name)
        value, ts = self._latest.get(sig_name, (spec.default, 0.0))
        age_ms = (time.time() - ts) * 1000.0 if ts else float("inf")
        if age_ms <= spec.stale_after_ms:
            return value
        if spec.stale_policy == "hold_last":
            return value
        if spec.stale_policy == "return_default":
            return spec.default
        return None

    # ------------------------------------------------ outbound command
    def send_command(self, name: str, args=None) -> Optional[str]:
        if name not in self._declared_commands:
            raise KeyError(f"{self.source_id}: undeclared command {name!r}")
        if self._socket is None:
            logger.warning("%s.send_command(%s): not bound", self.source_id, name)
            return None
        return self._socket.send_command(name, args or [])

class _ExternalSocket:
    """ROUTER socket for ONE ExternalHardware instance. Binds tcp://0.0.0.0:listen_port,
    accepts only DEALER frames whose identity == source_id, runs on a shared
    Tornado IOLoop (owned by the task). Dispatches typed messages back into
    the owning ExternalHardware via its _on_* hooks; rides Event_Dispatcher
    for EVT frames so they land in Elasticsearch on the existing CONTINUOUS
    pipe."""

    def __init__(self, owner, ioloop, event_dispatcher,
                 listen_port: int, source_id: str, stale_ms: int):
        self.owner = owner
        self.ioloop = ioloop
        self.event_dispatcher = event_dispatcher
        self.listen_port = int(listen_port)
        self.source_id = source_id
        self.source_id_bytes = source_id.encode("utf-8")
        self.stale_ms = int(stale_ms)

        self.last_seen_ms: float = 0.0
        self.alive: bool = False
        self.msg_count: int = 0
        self.dropped_count: int = 0
        self.type_mismatch_count: int = 0
        self.pending_commands: Dict[str, dict] = {}
        self._last_unknown_log_ms: float = 0.0
        self._last_typemismatch_log_ms: Dict[str, float] = {}
        self._incoming: collections.deque = collections.deque(maxlen=256)

        self._closed = False
        self._stream = None
        self._sock = None
        self._hb_tick = None

    # ------------------------------------------------ lifecycle
    def start(self) -> None:
        ctx = zmq.Context.instance()
        self._sock = ctx.socket(zmq.ROUTER)
        self._sock.bind(f"tcp://0.0.0.0:{self.listen_port}")
        self._stream = ZMQStream(self._sock, self.ioloop)
        self._stream.on_recv(self._handle_frame)
        self._hb_tick = PeriodicCallback(self._scan_staleness,
                                         max(100, self.stale_ms // 2))
        self._hb_tick.start()

    def close(self) -> None:
        if self._closed:
            return
        self._closed = True
        try:
            if self._hb_tick is not None:
                self._hb_tick.stop()
        except Exception:
            pass
        try:
            if self._stream is not None:
                self._stream.close()
        except Exception:
            pass

    # ------------------------------------------------ outbound
    def send_command(self, name: str, args: List[Any]) -> Optional[str]:
        if self._closed or self._stream is None:
            return None
        cmd_id = uuid.uuid4().hex
        payload = {"k": "CMD", "ts_pi": time.time(), "cmd_id": cmd_id,
                   "name": name, "args": list(args)}
        try:
            data = msgpack.packb(payload, use_bin_type=True)
            self._stream.send_multipart([self.source_id_bytes, data])
            self.pending_commands[cmd_id] = payload
            return cmd_id
        except Exception:
            logger.exception("%s: send_command(%s) failed", self.source_id, name)
            return None

    # ------------------------------------------------ inbound
    def _handle_frame(self, frames: List[bytes]) -> None:
        """ROUTER frames are [identity, payload]. Must NEVER raise."""
        try:
            if len(frames) < 2:
                return
            identity, payload = frames[0], frames[-1]
            if identity != self.source_id_bytes:
                self._rate_limited_warn(
                    f"{self.source_id}: rejected DEALER identity {identity!r}"
                )
                return
            try:
                msg = msgpack.unpackb(payload, raw=False)
            except Exception:
                self.dropped_count += 1
                self._rate_limited_warn(
                    f"{self.source_id}: malformed msgpack dropped"
                )
                return
            if not isinstance(msg, dict):
                self.dropped_count += 1
                return
            kind = msg.get("k")
            if kind not in _KINDS_INBOUND:
                self.dropped_count += 1
                return

            ts_pi_recv = time.time()
            self.msg_count += 1
            self.last_seen_ms = ts_pi_recv * 1000.0
            if len(self._incoming) >= self._incoming.maxlen:
                self.dropped_count += 1
            self._incoming.append((kind, msg, ts_pi_recv))

            if kind == "SIG":
                self._dispatch_sig(msg, ts_pi_recv)
            elif kind == "EVT":
                self._dispatch_evt(msg, ts_pi_recv)
            elif kind == "HB":
                self._mark_alive(True, ts_pi_recv)
            elif kind == "ACK":
                self._dispatch_ack(msg)
        except Exception:
            logger.exception("%s: _handle_frame failed", self.source_id)

    def _dispatch_sig(self, msg: dict, ts_pi_recv: float) -> None:
        sig = msg.get("sig")
        if not isinstance(sig, str):
            self.dropped_count += 1
            return
        spec = self.owner._declared_signals.get(sig)
        if spec is None:
            self._rate_limited_warn(
                f"{self.source_id}: unknown signal {sig!r} dropped"
            )
            self.dropped_count += 1
            return

        raw = msg.get("v")
        try:
            # bool is an int subclass — float(True) == 1.0 silently. Reject non-bool
            # raw values for bool signals so the contract is exact.
            if spec.dtype is bool:
                if not isinstance(raw, bool):
                    raise TypeError(f"expected bool, got {type(raw).__name__}")
                coerced = raw
            elif isinstance(raw, spec.dtype) and not (spec.dtype is float and isinstance(raw, bool)):
                coerced = raw
            else:
                coerced = spec.dtype(raw)
        except (TypeError, ValueError) as exc:
            self.type_mismatch_count += 1
            self.dropped_count += 1
            self._rate_limited_typemismatch(sig, exc)
            return

        self._mark_alive(True, ts_pi_recv)
        self.owner._on_signal_update(sig, coerced, ts_pi_recv)

    def _dispatch_evt(self, msg: dict, ts_pi_recv: float) -> None:
        evt = msg.get("evt")
        payload = msg.get("p", {}) or {}
        if not isinstance(evt, str) or evt not in self.owner._declared_events:
            self._rate_limited_warn(
                f"{self.source_id}: unknown event {evt!r} dropped"
            )
            self.dropped_count += 1
            return
        self._mark_alive(True, ts_pi_recv)
        try:
            self.owner._on_event(evt, payload, ts_pi_recv)
        except Exception:
            logger.exception("%s._on_event(%s) failed", self.source_id, evt)
        if self.event_dispatcher is not None:
            try:
                event_data = {
                    "func_name": evt,
                    "id": f"{self.source_id}.{evt}",
                    "value": payload,
                    "result": None,
                    "ts_src": msg.get("ts_src"),
                    "ts_pi_recv": ts_pi_recv,
                }
                ev = Event(event_type="ExternalEvent", event_data=event_data)
                self.event_dispatcher.dispatch_event(ev, key="CONTINUOUS")
            except Exception:
                logger.exception("%s: dispatch_event failed for %s",
                                 self.source_id, evt)

    def _dispatch_ack(self, msg: dict) -> None:
        cmd_id = msg.get("cmd_id")
        if cmd_id:
            self.pending_commands.pop(cmd_id, None)

    # ------------------------------------------------ liveness
    def _scan_staleness(self) -> None:
        try:
            now_ms = time.time() * 1000.0
            if self.alive and (now_ms - self.last_seen_ms) > self.stale_ms:
                self._mark_alive(False, now_ms / 1000.0)
        except Exception:
            logger.exception("%s: _scan_staleness failed", self.source_id)

    def _mark_alive(self, alive: bool, ts: float) -> None:
        if self.alive == alive:
            return
        self.alive = alive
        try:
            self.owner._on_stale_transition(alive)
        except Exception:
            logger.exception("%s: _on_stale_transition failed", self.source_id)
        if self.event_dispatcher is None:
            return
        try:
            ev = Event(
                event_type="ExternalSourceAlive",
                event_data={
                    "func_name": "alive_transition",
                    "id": f"{self.source_id}.alive",
                    "value": alive,
                    "result": None,
                    "ts_pi_recv": ts,
                },
            )
            self.event_dispatcher.dispatch_event(ev, key="CONTINUOUS")
        except Exception:
            logger.exception("%s: alive dispatch failed", self.source_id)

    # ------------------------------------------------ utils
    def _rate_limited_warn(self, line: str, min_gap_ms: float = 1000.0) -> None:
        now_ms = time.time() * 1000.0
        if now_ms - self._last_unknown_log_ms >= min_gap_ms:
            self._last_unknown_log_ms = now_ms
            logger.warning(line)

    def _rate_limited_typemismatch(self, sig: str, exc: Exception,
                                    min_gap_ms: float = 1000.0) -> None:
        """Rate-limit per-signal so a misbehaving sender can't flood logs."""
        now_ms = time.time() * 1000.0
        last = self._last_typemismatch_log_ms.get(sig, 0.0)
        if now_ms - last >= min_gap_ms:
            self._last_typemismatch_log_ms[sig] = now_ms
            logger.warning("%s.%s: type mismatch (%s) dropped",
                           self.source_id, sig, exc)

<edge_cases>
- **Test machine lacks pyzmq or msgpack** — same package set Net_Node uses; if missing, `pip install pyzmq msgpack` and rerun.
- **Port already in use** — verify scripts use high ports (15580/15585/15586) to avoid collision with the real EXTLINK ports a running task would bind. If those are also in use, pick another high port.
- **`View.__init__` requires a real Event_Dispatcher** — verify scripts construct one (no stub). DO NOT modify View to relax this.
- **Subclass overrides `_on_event` to do work** — keep base `_on_event` as a no-op; the default behaviour is "ride Event_Dispatcher CONTINUOUS and that's it". Authors who want side effects override it.
- **`stale_ms` is small (<200ms)** — `_hb_tick` interval is `max(100, stale_ms // 2)`. A pathological `stale_ms = 50` would scan every 100ms, mark stale after 50ms — possibly OK, possibly noisy. Don't enforce a lower bound; trust per-pilot config.
- **Tracker.set on a Boolean_Tracker with non-bool** — Boolean_Tracker.set raises on non-bool (Tracker.py:46). `_on_stale_transition` always passes a bool; `_on_signal_update` only touches non-boolean Trackers (signals are regular Tracker). No collision.
- **Re-bind on the same instance** — `.bind()` is idempotent: a second call is a no-op. Don't change ports between calls.
- **`bool` is an `int` subclass in Python** — `float(True) == 1.0` and `isinstance(True, int) is True`. `_dispatch_sig` special-cases `dtype is bool` to require `isinstance(raw, bool)`; conversely, for `dtype is float`, an incoming `True` would coerce to `1.0` silently. We also guard the int/float fast path against this: the `isinstance(raw, spec.dtype) and not (spec.dtype is float and isinstance(raw, bool))` clause forces bool-into-float through the explicit `spec.dtype(raw)` call — which still produces `1.0`, but at least the conversion is explicit and observable. If a downstream rig sends bools into a float signal accidentally, the cleanest fix is on the sender (Phase 19 SDK stub generation).
- **Annotation is a forward-ref string (`from __future__ import annotations`)** — `fn.__annotations__["return"]` is then `"float"` (a `str`, not the `float` type). `_resolve_signal_dtype` falls through to the `type(default)` branch. Authors who use future-annotations + no default get a clear `TypeError` at import time and either drop the future-annotation or add a typed default. Acceptable v1 limitation; documented.
- **Default type disagrees with annotation** — e.g. `@signal(default=0)` with `-> float` annotation. `_resolve_signal_dtype` picks annotation (`float`), then `signal()` coerces the default via `float(0) == 0.0`. Loud failure if coercion impossible (e.g. `default="x"` + `-> float`).
</edge_cases>
