def _decorator_name(node: ast.expr) -> str | None:
    """Return the bare name of a decorator. `@foo` → 'foo', `@foo(...)` → 'foo'."""
    if isinstance(node, ast.Name):
        return node.id
    if isinstance(node, ast.Call):
        return _decorator_name(node.func)
    if isinstance(node, ast.Attribute):
        return node.attr
    return None


def _decorator_kwargs(node: ast.expr) -> dict:
    """Pull literal kwargs from a decorator Call, e.g. @signal(default=0.0, ...)."""
    if not isinstance(node, ast.Call):
        return {}
    out: dict = {}
    for kw in node.keywords:
        if kw.arg is None:
            continue
        try:
            out[kw.arg] = ast.literal_eval(kw.value)
        except (ValueError, SyntaxError):
            out[kw.arg] = ast.unparse(kw.value)
    return out


def _annotation_to_dtype(node: ast.expr | None) -> str | None:
    """Map an annotation AST node to a string dtype name.
    `-> float` → 'float'; `-> None` → None; complex expr → its `ast.unparse` form.
    The Pi decorator enforces the actual restriction to {int, float, bool, str};
    this extractor is permissive so the UI can surface non-primitive annotations
    as errors rather than dropping them silently."""
    if node is None:
        return None
    if isinstance(node, ast.Name):
        return node.id
    if isinstance(node, ast.Constant) and node.value is None:
        return None
    try:
        return ast.unparse(node)
    except Exception:
        return None


def _default_literal_to_dtype(default_value) -> str | None:
    """Fallback: infer dtype from a literal default's Python type."""
    if default_value is None:
        return None
    t = type(default_value)
    if t in (int, float, bool, str):
        return t.__name__
    return None


def _command_args_from_funcdef(item: ast.FunctionDef) -> list[dict]:
    """Return [{name, dtype}, ...] for every non-self parameter of a @command."""
    out: list[dict] = []
    for arg in item.args.args:
        if arg.arg == "self":
            continue
        out.append({
            "name": arg.arg,
            "dtype": _annotation_to_dtype(arg.annotation),
        })
    return out


def _extract_extlink_decorators(item: ast.FunctionDef) -> tuple[
    dict | None, dict | None, dict | None
]:
    """Return (signal_spec, event_spec, command_spec) for this method, all None if none apply."""
    sig_spec: dict | None = None
    evt_spec: dict | None = None
    cmd_spec: dict | None = None
    for dec in item.decorator_list:
        name = _decorator_name(dec)
        if name == "signal":
            kw = _decorator_kwargs(dec)
            default = kw.get("default")
            # dtype resolution: return annotation first, then literal default's type.
            dtype = _annotation_to_dtype(item.returns) or _default_literal_to_dtype(default)
            sig_spec = {
                "name": item.name,
                "default": default,
                "stale_after_ms": kw.get("stale_after_ms", 200),
                "stale_policy": kw.get("stale_policy", "hold_last"),
                "dtype": dtype,   # may be None — UI must show as an error
            }
        elif name == "event":
            kw = _decorator_kwargs(dec)
            evt_spec = {
                "name": item.name,
                "payload": kw.get("payload", {}),
            }
        elif name == "command":
            cmd_spec = {
                "name": item.name,
                "args": _command_args_from_funcdef(item),
                "returns": _annotation_to_dtype(item.returns),
            }
    return sig_spec, evt_spec, cmd_spec

```python
def extract_ast_metadata(source_code: str) -> dict:
    """Parse Python source and return class/method structure for GUI and diffing.

    Existing shape: {classes: [{name, methods: [{name, args}]}]}.
    Phase 18 additive: each class may also carry an `extlink` sub-key
    {signals, events, commands} if any of the three MICS-Link decorators
    apply to its methods. Absent for classes that use none of them.
    """
    tree = ast.parse(source_code)
    classes = []
    for node in ast.walk(tree):
        if not isinstance(node, ast.ClassDef):
            continue
        methods = []
        signals: list[dict] = []
        events: list[dict] = []
        commands: list[dict] = []
        for item in node.body:
            if not isinstance(item, ast.FunctionDef):
                continue
            args = []
            for arg in item.args.args:
                if arg.arg == "self":
                    continue
                annotation = ast.unparse(arg.annotation) if arg.annotation else None
                args.append({"name": arg.arg, "annotation": annotation})
            defaults = item.args.defaults
            for i, default in enumerate(defaults):
                args[len(args) - len(defaults) + i]["default"] = ast.unparse(default)
            methods.append({"name": item.name, "args": args})

            sig_spec, evt_spec, cmd_spec = _extract_extlink_decorators(item)
            if sig_spec is not None:
                signals.append(sig_spec)
            if evt_spec is not None:
                events.append(evt_spec)
            if cmd_spec is not None:
                commands.append(cmd_spec)

        entry: dict = {"name": node.name, "methods": methods}
        if signals or events or commands:
            entry["extlink"] = {
                "signals": signals,
                "events": events,
                "commands": commands,
            }
        classes.append(entry)
    return {"classes": classes}
```

"""MICS-Link smoke test — RUN FROM THE DEV MACHINE, NOT THE PI.

Connects a DEALER socket to a single ExternalHardware ROUTER on a Pi and
exercises the wire protocol. The (--listen-port, --source-id) pair MUST match
the pilot_hardware_config row for that module on that pilot.

Subcommands:
  probe   — one-shot HB + SIG + EVT + a deliberately malformed frame
  sustain — sustained 10 Hz SIG + 1 Hz HB loop for N seconds
"""
import argparse
import logging
import sys
import time

try:
    import msgpack
    import zmq
except ImportError as e:
    print(f"ERROR: pyzmq + msgpack required (pip install pyzmq msgpack): {e}",
          file=sys.stderr)
    sys.exit(2)


def build_socket(pi_host: str, port: int, source_id: str):
    ctx = zmq.Context.instance()
    sock = ctx.socket(zmq.DEALER)
    sock.setsockopt_string(zmq.IDENTITY, source_id)
    sock.connect(f"tcp://{pi_host}:{port}")
    return ctx, sock


def send_sig(sock, sig: str, value, seq: int) -> None:
    sock.send(msgpack.packb(
        {"k": "SIG", "ts_src": time.time(), "seq": seq, "sig": sig, "v": value},
        use_bin_type=True))


def send_hb(sock, seq: int) -> None:
    sock.send(msgpack.packb(
        {"k": "HB", "ts_src": time.time(), "seq": seq}, use_bin_type=True))


def send_evt(sock, evt: str, payload: dict, seq: int) -> None:
    sock.send(msgpack.packb(
        {"k": "EVT", "ts_src": time.time(), "seq": seq, "evt": evt, "p": payload},
        use_bin_type=True))


def send_malformed(sock) -> None:
    sock.send(b"\xff\xff\xff\xff THIS IS NOT MSGPACK \xff")


def cmd_probe(args) -> int:
    ctx, sock = build_socket(args.pi_host, args.listen_port, args.source_id)
    try:
        print(f"[probe] connected as {args.source_id} → "
              f"{args.pi_host}:{args.listen_port}")
        send_hb(sock, seq=1)
        time.sleep(0.05)
        send_sig(sock, sig=args.signal, value=args.value, seq=2)
        time.sleep(0.05)
        send_evt(sock, evt=args.event,
                 payload={"object": "pellet", "confidence": 0.91}, seq=3)
        time.sleep(0.05)
        print(f"[probe] sent HB + SIG ({args.signal}={args.value}) + EVT ({args.event}).")

        send_malformed(sock)
        time.sleep(0.05)
        print("[probe] sent malformed frame (Pi must keep running).")

        send_hb(sock, seq=4)
        time.sleep(0.1)
        print("[probe] done. Tail the pilot log and look for tracker "
              f"updates on {args.source_id}.{args.signal}.")
    finally:
        sock.close(linger=200)
        ctx.term()
    return 0


def cmd_sustain(args) -> int:
    ctx, sock = build_socket(args.pi_host, args.listen_port, args.source_id)
    end_t = time.time() + args.seconds
    seq = 0
    last_hb = 0.0
    print(f"[sustain] {args.seconds}s @ 10 Hz SIG + 1 Hz HB → "
          f"{args.pi_host}:{args.listen_port}")
    try:
        while time.time() < end_t:
            now = time.time()
            seq += 1
            # Triangle wave so the value crosses any threshold inside [0, 1]
            phase = (now * 0.5) % 2
            value = phase if phase < 1 else 2 - phase
            send_sig(sock, sig=args.signal, value=value, seq=seq)
            if now - last_hb >= 1.0:
                seq += 1
                send_hb(sock, seq=seq)
                last_hb = now
            time.sleep(0.1)
        print("[sustain] done.")
    except KeyboardInterrupt:
        print("[sustain] interrupted.")
    finally:
        sock.close(linger=200)
        ctx.term()
    return 0


def main(argv=None) -> int:
    logging.basicConfig(level=logging.INFO,
                        format="%(asctime)s %(levelname)s %(message)s")

    p.add_argument("--pi-host", required=True,

    p.add_argument("--source-id", required=True,
                   help="DEALER identity = source_id from pilot_hardware_config.config")
    p.add_argument("--signal", default="left_paw_x",
                   help="Declared @signal name to push (default: left_paw_x)")
    p.add_argument("--event", default="object_detected",
                   help="Declared @event name to emit (default: object_detected)")
    p.add_argument("--value", type=float, default=0.7,
                   help="Value to push on the probe SIG (default: 0.7)")

    sub = p.add_subparsers(dest="mode", required=True)
    sp = sub.add_parser("probe", help="One-shot HB + SIG + EVT + malformed frame.")
    sp.set_defaults(func=cmd_probe)

    ss.add_argument("--seconds", type=float, default=15.0)

    return args.func(args)

<edge_cases>
- **AST extractor receives source with non-literal decorator kwargs** — e.g. `@signal(default=SOME_CONST, ...)`. `ast.literal_eval` raises `ValueError`; `_decorator_kwargs` falls back to `ast.unparse` (string form). Downstream UI will display the raw expression — acceptable, no crash.
- **Non-primitive annotations on @signal** — e.g. `-> Optional[float]` or `-> list[float]`. `_annotation_to_dtype` returns the unparsed string (`"Optional[float]"`); `dtype` in the AST metadata is a non-primitive string. The Pi-side decorator (Plan 18-01) is stricter and raises `TypeError` at import time. The UI consumes this as an error signal and refuses to wire the lib into a toolkit until the author narrows the annotation. Acceptable v1 behaviour — extractor is permissive (records what's there); decorator is strict (rejects non-primitives).
- **`dtype` resolves to null** — annotation absent AND default is not a primitive literal (e.g. `default=some_expression`). `dtype` is `None` in the AST shape; the Pi decorator will fail-loud at import. UI should surface `dtype == null` as an authoring error.
- **Command with `**kwargs` or `*args`** — `_command_args_from_funcdef` only walks `item.args.args` (positional/keyword). Variadic params are silently dropped from the captured signature. v1 contract is "commands have a fixed primitive signature"; document and revisit in Phase 19 if a real use case appears.
- **Gate timeout differs across required sources** — `max(hw.wait_timeout_s for hw in required_refs)` is used. The gate waits for the slowest tolerance, not the strictest. Rationale: we want enough time for ALL required sources to come online, not abort because the most-strict one timed out while a more-tolerant one was still booting.
- **Skip-wait arrives before the gate is armed** — `self._extlink_skip_wait = True` is set, but the gate either isn't installed (no required sources → noop) or hasn't been reached yet (FDA still constructing). When the FDA does reach `_wait_extlink_ready`, the very first transition evaluation finds `_gate_open() → True` and exits immediately. Acceptable — a researcher who pre-emptively requested skip gets the behavior they asked for.
- **Skip-wait arrives after the gate has already released** — `_extlink_skip_wait` is set to True; the FDA is no longer in `_wait_extlink_ready`, so no transition fires. Handler logs `idempotent=True`. CONTINUOUS audit event still emitted (useful trail of "operator pressed skip even though it wasn't needed").
- **Required source alive, then dies, before gate releases** — `hw.alive` is monitored continuously by the stale scan. If a required source goes alive briefly then stale before gate-open is evaluated on that FDA tick, the gate stays closed. Once all required sources are alive on the *same FDA tick*, the transition fires. Acceptable — researchers can use `stale_ms` to tune this.
- **Two required sources, one comes alive, the other never does** — gate waits until BOTH alive, OR timeout, OR manual skip. Timeout is `max(wait_timeout_s)` across both. The pilot log lists the still-offline source(s) every time `_wait_extlink_ready` is entered.
- **Task with NO required externals** — `self._required_externals = []`; the `if self._required_externals:` branch in `load_fda_from_json` is skipped entirely; no synthetic state is added; the user's initial state is the FDA's initial state. Researchers running non-MICS-Link tasks see zero behavior change and zero overhead. Critical backward-compat property.
- **Orchestrator offline when researcher needs to skip** — the API endpoint is unreachable, the curl POST fails. The researcher can still abort via STOP (which uses a different ZMQ path). Worst case, they wait for timeout. All three escape paths are independent; losing one does not lock the system.
- **Pilot offline when skip POST arrives** — endpoint returns 409, no message sent. Researcher must wait for timeout or kill the orchestrator+pilot combo manually. Acceptable since the run is dead anyway.
- **Source has `@signal` from a different module** — the extractor matches by bare decorator name. If someone imports an unrelated `signal` function as `signal`, we'd misclassify. Acceptable for v1; document in Phase 19 if it becomes a problem.
- **Pi can't find pyzmq + msgpack** — same packages Net_Node uses; if a fresh pi-mirror lacks them, the user installs via the standard pilot setup script. The smoke script gives a clear ERROR + exit 2 if its dev-machine env is missing them.
- **Multiple pilots on the same external module** — each pilot has its own `pilot_hardware_config` row with its own `listen_port`. No collision unless the user reuses a port across pilots on the same physical Pi (not a possibility — only one Pi per pilot).
- **Port already bound (zombie task)** — `_ExternalSocket.start` will raise on bind; `init_hardware` post-pass catches + logs and the instance stays unbound. User must kill the stale process or pick another port in `pilot_hardware_config`. The base `task.end()` loop calls `release()` so a clean shutdown frees the port; only crashes leak it.
- **Task ends while a CMD ACK is in flight** — the IOLoop stops; `_stream` is closed; the SDK sees a disconnect and stops waiting. Acceptable; no Pi-side cleanup needed beyond the existing `close()`.
- **IOLoop in daemon thread keeps running on process exit** — daemon=True ensures process termination doesn't block. Pilot graceful shutdown calls `mics_task.end()` which stops the loop explicitly anyway.
- **`mics_task.end()` raises before `_extlink_ioloop.stop`** — the `try/finally` wrapper ensures the stop still runs; otherwise the daemon thread would outlive the task object until process exit.
</edge_cases>
