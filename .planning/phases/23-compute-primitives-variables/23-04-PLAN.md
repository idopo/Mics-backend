def _validate_output_spec(self, action):
    spec = action.get("output")
    if spec is None: return
    for name in ([spec] if isinstance(spec, str) else list(spec)):
        if name not in self.flags: raise ValueError(...)

def _capture_output(self, action, result):
    for name, value in unpack_output(action.get("output"), result):
        self.flags[name].set(value)         # <-- this is the CMP-16 result-level Tracker event
