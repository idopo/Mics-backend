```python
# api/device_lease.py   (NEW, plan 18-08)
ROLE_ROUTER_BIND = "router_bind"
ROLE_SUB_CONNECT = "sub_connect"

def normalize_host(host: str) -> str:
    """Lease key derivation (EXTLINK-17). Strips whitespace, lowercases, strips a scheme
       (`http://`, `tcp://`), strips a trailing `/`, and DROPS any `:port` suffix — one
       physical box = one lease no matter how many ports or modules target it."""

def is_extlink_config(cfg: dict) -> bool:
    """True iff cfg["role"] is one of the two known roles."""

def validate_extlink_config(module_name: str, module_id, cfg: dict) -> list[dict]:
    """EXTLINK-10/13 field validation, emitted as `extlink_config_invalid` issues:
       unknown role; sub_connect without host/connect_port; router_bind without listen_port;
       missing source_id; wait_timeout_s null / non-int / outside [5, 600];
       stale_ms non-positive; egress_fail_threshold non-positive."""

def device_held_issue(module_name: str, module_id, host: str, holder: dict) -> dict:
    """{"module_id", "module_name", "issue": "device_held", "host", "holder": {...},
        "detail": "<host> is held by pilot '<p>' (subject <s>, run <r>) since <ts>"}"""

def preflight_device_lease_issues(db, pilot_id: int, modules: list, configs: dict) -> list[dict]:
    """One `device_held` issue per module whose normalized host is leased by a DIFFERENT pilot."""

def get_lease(db, host: str) -> dict | None
def acquire_lease(db, host, pilot_id, pilot_name, session_id, run_id, subject_key) -> tuple[bool, dict | None]
def release_leases_for_run(db, run_id: int) -> list[str]
def force_release(db, host: str) -> bool
def reconcile_leases(db, heartbeats: dict, now=None, stale_after_s: int = 90) -> list[dict]:
    """`heartbeats` maps pilot_name -> ISO-8601 `updated_at` (the orchestrator's Redis
       `_redis_touch` value). A lease whose holder is ABSENT from the map, or whose timestamp
       is older than stale_after_s, is released. Pure over the injected map — never reads
       Redis itself, which is what makes it unit-testable with a fabricated stale timestamp."""
```
