# ZERO `autopilot` imports at module scope. stdlib + msgpack only.

WIRE_KINDS = ("SIG", "EVT", "HB", "ACK", "CMD")
ALLOWED_DTYPES = {"int": int, "float": float, "bool": bool, "str": str}

ROLE_ROUTER_BIND = "router_bind"
ROLE_SUB_CONNECT = "sub_connect"

@dataclass
class SignalSpec:
    name: str
    dtype: type
    default: Any
    stale_after_ms: int          # 0 == never stale
    stale_policy: str            # "hold_last" | "return_default" | "return_none"

@dataclass
class DecodeStats:
    malformed: int = 0           # bad msgpack / non-dict / unknown kind / decoder raised
    unknown_name: int = 0        # sig/evt name not declared by the class (EXTLINK-08)
    type_mismatch: int = 0       # dtype coercion failed (EXTLINK-12)

def resolve_dtype(annotation: Any, default: Any) -> type:
    """Annotation first, then type(default). TypeError if neither yields a member of
       ALLOWED_DTYPES' values. Raised at class-build time (EXTLINK-12)."""

def encode(kind: str, **fields) -> bytes:
    """msgpack.packb({"k": kind, **fields}, use_bin_type=True).
       ValueError naming `kind` if kind not in WIRE_KINDS."""

def decode_envelope(raw: bytes, stats: DecodeStats) -> dict | None:
    """NEVER raises. Returns None and increments stats.malformed on: msgpack failure,
       non-dict payload, missing "k", or "k" not in WIRE_KINDS."""

def coerce_value(raw: Any, dtype: type, stats: DecodeStats) -> tuple[bool, Any]:
    """(ok, value). NEVER raises.
       bool is special-cased: only isinstance(raw, bool) is accepted — float(True) must NOT
       silently pass as 1.0 (Python's bool is an int subclass).
       On TypeError/ValueError: stats.type_mismatch += 1 -> (False, None)."""

def apply_envelope(env: dict,
                   signals: dict[str, SignalSpec],
                   events: set[str],
                   stats: DecodeStats) -> list[tuple]:
    """Validate a decoded envelope against the class's declared names. NEVER raises.
       Returns a list of update tuples:
         ("sig", name, value, ts_src, seq)
         ("evt", name, payload, ts_src, seq)
         ("hb",  None, None, ts_src, seq)
         ("ack", cmd_id, result, ts_src, None)
       An undeclared sig/evt name -> stats.unknown_name += 1 and NO tuple (EXTLINK-08).
       A dtype-coercion failure -> stats.type_mismatch += 1 and NO tuple."""

def resolve_stale_value(spec: SignalSpec, last_value: Any,
                        last_ts_ms: float | None, now_ms: float) -> tuple[bool, Any]:
    """(should_write, value) for the per-signal stale policy (EXTLINK-06).
         fresh (or stale_after_ms == 0)  -> (False, last_value)
         stale + "hold_last"             -> (False, last_value)
         stale + "return_default"        -> (True, spec.default)
         stale + "return_none"           -> (True, None)
       last_ts_ms is None (nothing ever received) counts as stale."""

def default_liveness(last_msg_ts_ms: float | None, now_ms: float, stale_ms: int) -> bool:
    """EXTLINK-07 base default: any successfully decoded message within stale_ms.
       last_msg_ts_ms None -> False."""

def make_liveness(predicate=None):
    """Returns a callable(last_msg_ts_ms, now_ms, stale_ms) -> bool.
       A lib-supplied predicate REPLACES the default entirely — the data-arrival check must
       never be consulted when an override is present."""

def socket_plan(config: dict) -> dict:
    """Pure role -> socket plan. No zmq import, no socket created.
       router_bind -> {"socket_type": "ROUTER", "action": "bind",
                       "endpoint": "tcp://0.0.0.0:{listen_port}",
                       "identity": source_id, "check_identity": True, "subscribe": None}
       sub_connect -> {"socket_type": "SUB", "action": "connect",
                       "endpoint": "tcp://{host}:{connect_port}",
                       "identity": None, "check_identity": False, "subscribe": ""}
       ValueError naming the missing/unknown field otherwise (unknown role, absent
       listen_port / host / connect_port)."""

def identity_ok(frame_identity, source_id: str) -> bool:
    """bytes or str accepted; bytes decoded utf-8, errors -> False. Never raises."""

def run_decoder(decoder_fn, frames: list, signals, events, stats: DecodeStats) -> list[tuple]:
    """Call a lib's @decoder on a foreign frame list. NEVER raises — any exception inside
       decoder_fn increments stats.malformed and returns [].
       decoder_fn may return: a dict {name: value}, a list of (name, value) pairs, or None.
       Each produced (name, value) is routed through the SAME declared-name check and dtype
       coercion apply_envelope uses, producing the same ("sig"/"evt", ...) tuples with
       ts_src=None, seq=None."""
