```python
def _build_tree_lambda(self, node: dict):
    """
    Recursively build a zero-argument callable from a condition_tree node.

    Leaf node (dict with 'left' key or legacy 'view' key):
        delegates to _build_transition_lambda(node).
    AND-branch (op='AND', children=[...]):
        returns lambda: all(child() for child in child_fns)
    OR-branch (op='OR', children=[...]):
        returns lambda: any(child() for child in child_fns)

    Default-arg capture used throughout to avoid late-binding closure issues.
    """
    if "op" not in node or node["op"] not in ("AND", "OR"):
        # Leaf node — has 'left'/'op'/'right' (unified) or 'view'/'op'/'rhs' (legacy)
        return self._build_transition_lambda(node)

    children = node.get("children", [])
    if not children:
        # Empty branch — unconditional (truthy)
        return lambda: True

    child_fns = [self._build_tree_lambda(child) for child in children]

    if node["op"] == "AND":
        def _and_check(fns=child_fns):
            return all(f() for f in fns)
        return _and_check
    else:  # OR
        def _or_check(fns=child_fns):
            return any(f() for f in fns)
        return _or_check
```
