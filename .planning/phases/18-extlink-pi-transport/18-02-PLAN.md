class EgressWorker:
    """One FIFO worker thread per device (EXTLINK-15). Fire-and-forget, NO retry."""
    def __init__(self, send_fn, maxsize=64, fail_threshold=3,
                 on_drop=None, on_alive_change=None, name=""): ...

    consecutive_failures: int
    alive: bool                  # starts True
    def enqueue(self, item) -> bool:
        """put_nowait. On queue.Full: dropped += 1, on_drop(item) called, returns False.
           NEVER blocks the caller. NEVER raises."""
    def stop(self, timeout=2.0) -> None:
        """Sentinel shutdown, joins the thread. Idempotent."""
    def drain(self, timeout=2.0) -> bool:
        """Test/teardown helper: block until the queue is empty or timeout. Returns True if empty."""

# Semantics the tests pin:
#  * FIFO under a single worker.
#  * queue.Full drops the NEWEST (stdlib put_nowait behaviour) — items already queued still send.
#  * send_fn raising is swallowed; the worker survives and attempts the NEXT item (no retry).
#  * on_alive_change(False) fires EXACTLY ONCE on the Nth consecutive failure (edge-triggered).
#  * on_alive_change(True) fires exactly once when a send succeeds after a failing streak.

RUN_CTX_KEYS = ("run_id", "session_id", "subject_key", "pilot", "task_definition_id", "started_at")

    ready: bool
    last_error: BaseException | None

    def join(self, timeout=None) -> None
