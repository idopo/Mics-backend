```python
def _report_trigger_error(self, pin, trig, exc):
    # Total by construction: this runs inside execute_trigger's worker thread, and anything
    # that escapes it kills trigger processing for the rest of the run (process_queue has no
    # handler — STATE.md Outstanding item 4, deliberately still open).
    try:
        name = getattr(trig, "__name__", repr(trig))
        self.logger.error(
            f"Trigger callback {name} for {pin} raised {type(exc).__name__}: {exc}",
            exc_info=True,
        )
        dispatcher = getattr(self, "event_dispatcher", None)
        if dispatcher is None:
            return
        dispatcher.dispatch_event(Event(
            event_type="TRIGGER_ACTION_ERROR",
            event_data={"trigger": str(pin), "callback": name,
                        "error_type": type(exc).__name__, "error": str(exc)},
        ))
    except Exception:
        # Last resort: never let the error path raise. Swallow, and do not re-log via
        # self.logger (which may be what failed).
        pass
```
