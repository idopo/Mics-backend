## Step 2: Auto-migrate in normaliseTransition()

**File:** `web_ui/react-src/src/pages/task-editor/TaskEditor.tsx`

`normaliseTransition()` (lines 34–50) currently resolves `from`, `to`, and `conditions[]` from
various legacy shapes. Extend it to always produce `condition_groups`:

```typescript
function normaliseTransition(t: any): FdaTransition {
  const from: string = t.from ?? t.from_state ?? ''
  const to: string   = t.to   ?? t.next_state  ?? ''

  // Resolve legacy conditions[] (one or zero conditions)
  let legacyConditions: FdaCondition[] = t.conditions ?? []
  if (legacyConditions.length === 0 && t.condition) {
    const c = t.condition
    if ('left' in c) {
      legacyConditions = [c]
    } else {
      legacyConditions = [{ left: { view: c.view ?? '' }, op: c.op ?? '==', right: c.rhs ?? 0 }]
    }
  }

  // If already has condition_groups, use as-is (already migrated).
  // Empty legacyConditions → [] (not [{conditions:[]}]) so ConditionGroupsEditor shows
  // the "unconditional" hint instead of an empty group card.
  const condition_groups: ConditionGroup[] = t.condition_groups
    ?? (legacyConditions.length === 0 ? [] : [{ conditions: legacyConditions }])

  return { from, to, condition_groups, description: t.description }
}
```

After this, every `FdaTransition` in the editor always carries `condition_groups`. The legacy
`conditions` field is dropped from in-memory state (but the Pi still reads it from old stored JSON
via the fallback in Step 5).

## Step 3: Update condLabel()

**File:** `web_ui/react-src/src/pages/task-editor/TaskEditor.tsx`

Replace the existing `condLabel()` function (currently uses `t.conditions[0]`):

```typescript
function condLabel(t: FdaTransition): string {
  const groups = t.condition_groups ?? []
  if (groups.length === 0 || groups.every(g => g.conditions.length === 0)) {
    return '(unconditional)'
  }
  return groups
    .map(g =>
      g.conditions
        .map(c => `${operandLabel(c.left)} ${c.op} ${operandLabel(c.right)}`)
        .join(' ∧ ')
    )
    .join(' ∨ ')
}
```

Renders as:
- Single condition: `lick_port == true`
- AND group: `lick_port == true ∧ trial_counter >= 5`
- OR groups: `lick_port == true ∧ trial_counter >= 5 ∨ timeout_expired == true`

const updateTransitionGroups = (edgeId: string, groups: ConditionGroup[]) => {
  if (!fdaJson) return
  const idx = parseInt(edgeId.replace('e-', ''), 10)
  if (isNaN(idx) || idx < 0 || idx >= fdaJson.transitions.length) return
  const newTransitions = fdaJson.transitions.map((t, i) =>

  setFdaJson(prev => prev ? { ...prev, transitions: newTransitions } : prev)

  setEdges(eds => eds.map(e => e.id === edgeId ? { ...e, label } : e))
}

## Step 5: Pi — update load_fda_from_json() transition loop

**File:** `~/pi-mirror/autopilot/autopilot/tasks/mics_task.py`

> **⚠️ Correction from original plan:** `_build_transition_lambda(self, cond: dict)` takes a
> **single condition leaf** (called once per condition in a loop at line 819). It must NOT be
> changed. The DNF logic goes in the **caller** — the transition registration loop starting at
> line 803 inside `load_fda_from_json()`.

Replace the transition registration block (lines 803–826) with:

```python
# ── 5. Register transitions ──────────────────────────────────────────────────────
for trans in definition.get("transitions", []):
    from_name = trans["from"]
    to_name   = trans["to"]
    description = trans.get("description", "")

    if from_name not in state_method_map:
        raise ValueError(
            f"load_fda_from_json: transition 'from' state '{from_name}' not in states dict."
        )
    if to_name not in state_method_map:
        raise ValueError(
            f"load_fda_from_json: transition 'to' state '{to_name}' not in states dict."
        )

    condition_groups = trans.get("condition_groups")

    if condition_groups is not None:
        # DNF path: OR of AND-groups
        if not condition_groups or any(
            len(g.get("conditions", [])) == 0 for g in condition_groups
        ):
            # Empty groups = unconditional
            expr_list = []
        else:
            group_fns = [
                [self._build_transition_lambda(c) for c in g.get("conditions", [])]
                for g in condition_groups
            ]
            # Single callable encoding any(all(...)) — avoids late-binding via default arg
            def _dnf(gfns=group_fns):
                return any(all(c() for c in g) for g in gfns)
            expr_list = [_dnf]
    else:
        # Legacy path: flat conditions[] — each lambda is ANDed by FDA.__next__
        conditions = trans.get("conditions", [])
        expr_list = [self._build_transition_lambda(cond) for cond in conditions]

    self.stages.add_transition(
        state_method_map[from_name],
        state_method_map[to_name],
        expr_list,
        trans_description=description,
    )
```

`_build_transition_lambda()` (leaf-condition builder, lines 931–973) is **unchanged**.

> **FiniteDeterministicAutomaton.py is unchanged.** `__next__` does `all(expr() for expr in expr_list)`;
> for the DNF path we pass a single-element list `[_dnf]` so this still works correctly.
> `boolean_conflict()` may under-warn on DNF-overlapping transitions, but this is advisory-only.
