```typescript
/** Recursive condition tree node. Leaf = a single FdaCondition; branch = AND/OR of children. */
export type ConditionNode =
  | FdaCondition                                          // leaf: has left/op/right
  | { op: 'AND' | 'OR'; children: ConditionNode[] }      // branch

/** Type guard: is this a branch node (not a leaf FdaCondition)? */
export function isConditionBranch(node: ConditionNode): node is { op: 'AND' | 'OR'; children: ConditionNode[] } {
  return typeof node === 'object' && node !== null && ('op' in node) && (node.op === 'AND' || node.op === 'OR') && 'children' in node
}
```

**New props interface:**
```typescript
interface Props {
  tree: ConditionNode | null        // null = empty (unconditional)
  toolkit: ToolkitRead | null
  hwModuleNames?: string[]
  onChange: (tree: ConditionNode | null) => void
}
```

**Full rewrite spec:**

1. Import `React` (for `React.Fragment`), `isConditionBranch`, `ConditionNode`, `FdaCondition`, `ToolkitRead` from types. Import `ConditionRow` from `./ConditionBuilder`.

2. `EMPTY_LEAF: FdaCondition = { left: { view: '' }, op: '==', right: 0 }` — sentinel for new conditions.

3. **Tree mutation helpers** (pure functions — no React state):

   `addAndSibling(tree, path)`: Given a `path` (array of indices into children arrays) pointing to a node, insert a new `EMPTY_LEAF` as the next sibling in the AND-context at that path. The AND-context is the nearest AND-branch ancestor (or the root if root is AND). If the target node's parent is an AND-branch, append EMPTY_LEAF after the target. If parent is an OR-branch, wrap the target + new leaf in a new AND-branch, replacing the target.

   `addOrSibling(tree, path)`: Insert a new `EMPTY_LEAF` as a sibling at the OR level. Walk up from path to find the nearest OR-ancestor (or root if root is OR). Append EMPTY_LEAF after the subtree. If no OR-ancestor exists, wrap the entire tree in an OR-branch with a new EMPTY_LEAF.

   `updateNode(tree, path, updated)`: Replace the node at `path` with `updated`. Returns new tree.

   `deleteNode(tree, path)`: Remove the node at `path`. After deletion, collapse: if a branch has 0 children → remove it (return null for root or remove from parent); if a branch has 1 child → replace the branch with its child.

4. **Recursive renderer `renderNode(node, path, depth)`:**

   ```
   Background opacity per depth:
   depth=0: no extra background (transparent)
   depth=1: rgba(255,255,255,0.03)
   depth=2: rgba(255,255,255,0.055)
   depth=3+: rgba(255,255,255,0.08)
   ```

   - If `isConditionBranch(node)`:
     - Render a `div` with: `border: '1px solid var(--border)'`, `borderRadius: 4`, `padding: '6px 8px'`, `marginBottom: 4`, `background: <depth-opacity>`
     - Between children, render a small label: `AND` or `OR` (matching node.op) in `font-size:10px, color:var(--accent), fontWeight:700, letterSpacing:1, padding:'2px 0'`
     - Each child: `renderNode(child, [...path, i], depth + 1)` 

   - If leaf (FdaCondition):
     - Render `ConditionRow` with:
       - `condition={node}`
       - `toolkit`, `hwModuleNames`
       - `onChange={updated => onChange(updateNode(tree!, path, updated))}`
       - `onDelete={() => onChange(deleteNode(tree!, path))}` — pass null to parent if tree becomes null
     - Below the row (or inline after ConditionRow), render the +AND / +OR buttons:
       ```
       style: background:none; border:none; color:var(--accent); cursor:pointer; fontSize:11px; padding:'2px 6px'; marginRight:4px
       ```
       +AND: `onClick={() => onChange(addAndSibling(tree!, path))}`
       +OR: `onClick={() => onChange(addOrSibling(tree!, path))}`

5. **Root render:** When `tree === null`, show: 
   ```
   <div style={{ fontSize:11, color:'var(--text-muted)', padding:'2px 0 6px', fontStyle:'italic' }}>
     unconditional — fires immediately
   </div>
   ```
   Plus a single "+ Add condition" button that calls `onChange(EMPTY_LEAF)`.

   When `tree !== null`, render `renderNode(tree, [], 0)`.

6. Export as `export function ConditionGroupsEditor(...)` — same name as before.

**Critical correctness rule:** The tree mutation helpers MUST use default-arg capture or closures that don't rely on late binding. Use pure functional transforms (spread, filter, map) — no in-place mutation.
